\[ S_N=\frac{W_N}{1+e^{-\alpha\,[\log_{10}N-\log_{10}R_N]}},\qquad W_N=40,\;\alpha=2,\;R_N=2000 \]

\[ S_D=W_D\left(1-e^{-\beta D}\right),\qquad W_D=10,\;\beta=0{,}001 \]

\[ S_L=W_L\,\max\!\left[0,\;\frac{50-L}{50}\right],\qquad W_L=10 \]

\[ S_I=\sum_{v=1}^{V}p_v\,w_v,\qquad \omega_v=\sigma_1|x_v|+\sigma_2|y_v|,\qquad w_v=W_I\,\frac{\omega_v}{\sum_u \omega_u} \]

\[ UGI=S_N+S_D+S_L+S_I \]
